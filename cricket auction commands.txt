import asyncio
from sqlalchemy import update
from app.core.database import AsyncSessionLocal
from app.models.auction import AuctionEvent, AuctionStatus

async def make_draft():
    async with AsyncSessionLocal() as db:
        await db.execute(
            update(AuctionEvent)
            .where(AuctionEvent.id == 2)
            .values(status=AuctionStatus.draft)
        )
        await db.commit()
        print('Event #2 status reset to draft!')

asyncio.run(make_draft())
